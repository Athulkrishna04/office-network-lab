! ------------------------------------------------------------------ [M1] core
configure terminal
hostname SW1
no ip domain-lookup
line con 0
 logging synchronous
! Transparent: VLANs are created on purpose on each switch, never learned from VTP.
vtp mode transparent
!
vlan 10
 name SALES
vlan 20
 name HR
vlan 30
 name IT
vlan 40
 name GUEST
!
! Same port blocks on every access switch
interface range FastEthernet0/1 - 6
 description SALES access
 switchport mode access
 switchport access vlan 10
interface range FastEthernet0/7 - 12
 description HR access
 switchport mode access
 switchport access vlan 20
interface range FastEthernet0/13 - 18
 description IT access
 switchport mode access
 switchport access vlan 30
interface range FastEthernet0/19 - 20
 description GUEST AP uplink
 switchport mode access
 switchport access vlan 40
!
! Explicit allowed list: fault F2 depends on it
interface GigabitEthernet0/1
 description TRUNK to R1 G0/0
 switchport mode trunk
 switchport trunk allowed vlan 10,20,30,40
interface GigabitEthernet0/2
 description TRUNK to SW2 Gi0/1
 switchport mode trunk
 switchport trunk allowed vlan 10,20,30,40
!
! Management lives in the IT VLAN
interface Vlan1
 shutdown
interface Vlan30
 description Management
 ip address 192.168.10.131 255.255.255.192
 no shutdown
ip default-gateway 192.168.10.129
end
write memory

! ------------------------------------------------------------------ [M2] port security
configure terminal
! Wired users + server: one known device per port, remembered, port shuts on violation
interface range FastEthernet0/1 - 18
 switchport port-security
 switchport port-security maximum 1
 switchport port-security mac-address sticky
 switchport port-security violation shutdown
! AP uplink: every wireless client's MAC appears here, so allow many and don't sticky
interface range FastEthernet0/19 - 20
 switchport port-security
 switchport port-security maximum 10
 switchport port-security violation restrict
end
! Generate traffic from every host (ipconfig /renew) BEFORE saving, so the sticky MACs are saved too.
write memory
