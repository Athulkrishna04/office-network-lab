! =============================================================================
!  ISP | Simulated internet provider | Cisco 2911 in Packet Tracer
! -----------------------------------------------------------------------------
!  G0/0 -> R1 G0/1 (203.0.113.0/30) · G0/1 -> WEB-EXT (198.51.100.0/24)
!  Addresses come from the RFC 5737 documentation ranges, so they can never
!  collide with a real network.
! =============================================================================

! ------------------------------------------------------------------ [M2]
configure terminal
hostname ISP
no ip domain-lookup
line con 0
 logging synchronous
!
interface GigabitEthernet0/0
 description Customer link to R1 G0/1
 ip address 203.0.113.1 255.255.255.252
 no shutdown
!
interface GigabitEthernet0/1
 description Internet segment - WEB-EXT
 ip address 198.51.100.1 255.255.255.0
 no shutdown
!
! There is deliberately NO route to 192.168.10.0/24. Private space isn't routed
! on the internet, so replies only get back because R1 translates (PAT) them to
! 203.0.113.2. Test T14 checks this.
end
write memory
