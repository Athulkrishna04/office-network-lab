!  R1 | Edge router + router-on-a-stick | Cisco 2911 in Packet Tracer
! -----------------------------------------------------------------------------
!  How to use: at the Router> prompt type `enable`, then paste ONE section at a
!  time. Each section enters config mode and saves when it finishes.
!  Sections: [M1] core · [M2] WAN + PAT · [M2] guest isolation · [E1] hardening
!  Using an ISR4331? Rename G0/0 -> G0/0/0 and G0/1 -> G0/0/1.
!  STARTER CONFIG: in phase P5, replace this file with the real exported
!  running-config.
! =============================================================================

! ------------------------------------------------------------------ [M1] core
configure terminal
hostname R1
no ip domain-lookup
line con 0
 logging synchronous
!
interface GigabitEthernet0/0
 description TRUNK to SW1 Gi0/1
 no ip address
 no shutdown
!
interface GigabitEthernet0/0.10
 description VLAN10 SALES gateway
 encapsulation dot1Q 10
 ip address 192.168.10.1 255.255.255.192
 ip helper-address 192.168.10.130
!
interface GigabitEthernet0/0.20
 description VLAN20 HR gateway
 encapsulation dot1Q 20
 ip address 192.168.10.65 255.255.255.192
 ip helper-address 192.168.10.130
!
! No helper on .30: SRV-CORE is in VLAN 30 and hears those broadcasts directly.
interface GigabitEthernet0/0.30
 description VLAN30 IT gateway - SRV-CORE is local
 encapsulation dot1Q 30
 ip address 192.168.10.129 255.255.255.192
!
interface GigabitEthernet0/0.40
 description VLAN40 GUEST gateway
 encapsulation dot1Q 40
 ip address 192.168.10.193 255.255.255.192
 ip helper-address 192.168.10.130
end
write memory

! ------------------------------------------------------------------ [M2] WAN + PAT
configure terminal
interface GigabitEthernet0/1
 description WAN to ISP G0/0
 ip address 203.0.113.2 255.255.255.252
 ip nat outside
 no shutdown
!
interface GigabitEthernet0/0.10
 ip nat inside
interface GigabitEthernet0/0.20
 ip nat inside
interface GigabitEthernet0/0.30
 ip nat inside
interface GigabitEthernet0/0.40
 ip nat inside
!
ip route 0.0.0.0 0.0.0.0 203.0.113.1
!
access-list 1 permit 192.168.10.0 0.0.0.255
ip nat inside source list 1 interface GigabitEthernet0/1 overload
end
write memory

! ------------------------------------------------------------------ [M2] guest isolation
! Policy: guests may use DHCP, and DNS on SRV-CORE; they may not reach Sales, HR
! or IT; everything else (the internet) is allowed.
! First match wins. An invisible "deny ip any any" follows line 60.
! If your PT build rejects the leading sequence numbers, leave them out: IOS
! numbers the entries 10, 20, 30 ... on its own.
configure terminal
ip access-list extended GUEST-IN
 10 permit udp any any eq 67
 20 permit udp 192.168.10.192 0.0.0.63 host 192.168.10.130 eq 53
 30 deny ip 192.168.10.192 0.0.0.63 192.168.10.0 0.0.0.63
 40 deny ip 192.168.10.192 0.0.0.63 192.168.10.64 0.0.0.63
 50 deny ip 192.168.10.192 0.0.0.63 192.168.10.128 0.0.0.63
 60 permit ip 192.168.10.192 0.0.0.63 any
!
interface GigabitEthernet0/0.40
 ip access-group GUEST-IN in
end
write memory

! ------------------------------------------------------------------ [E1] hardening (after v1.0)
! LAB-ONLY placeholders. Never put a real password in a public repo.
! Do R1 before the switches so the native VLAN matches on both ends.
configure terminal
enable secret LAB-ONLY-change-me
service password-encryption
banner motd # Authorized access only - lab device #
ip domain-name office.test
username netadmin secret LAB-ONLY-change-me
! If PT rejects the one-liner, run "crypto key generate rsa" and answer 1024.
crypto key generate rsa general-keys modulus 1024
ip ssh version 2
access-list 10 permit 192.168.10.128 0.0.0.63
line vty 0 4
 login local
 transport input ssh
 access-class 10 in
line con 0
 exec-timeout 10 0
!
interface GigabitEthernet0/0.999
 description Native VLAN - untagged, no IP
 encapsulation dot1Q 999 native
end
write memory
